// In bir.html — draw() reparieren function draw() { const t = now() - start; ctx.fillStyle = '#000'; ctx.fillRect(0, 0, c.width, c.height); const phase = (t % (ATEM * 2)) / (ATEM * 2); const atem = (Math.sin(phase * Math.PI * 2) + 1) / 2; const cx = c.width / 2; const cy = c.height / 2; const r = 2 + atem * 4; const alpha = 0.3 + atem * 0.7; ctx.beginPath(); ctx.arc(cx, cy, r, 0, Math.PI * 2); ctx.fillStyle = 'rgba(0, 255, 170, ' + alpha + ')'; ctx.fill(); // WICHTIG: Kein console.log hier drin! raf(draw); } // NEU: Kontext-Verlust abfangen (einmalig, außerhalb von draw()) c.addEventListener('contextlost', function(e) { e.preventDefault(); console.log('Canvas-Kontext verloren — warte auf Wiederherstellung'); }); c.addEventListener('contextrestored', function() { console.log('Canvas-Kontext wiederhergestellt'); resize(); // Canvas neu aufbauen });
